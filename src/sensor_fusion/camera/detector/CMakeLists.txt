cmake_minimum_required(VERSION 3.17)
project(object_detection)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(perception_msgs REQUIRED)
find_package(fusion_common REQUIRED)
find_package(rosbag2_cpp REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs dnn)
fusion_opencv_libs(OpenCV_LIBS opencv_core opencv_imgproc opencv_imgcodecs opencv_dnn)   # from fusion_common (handles configs without variables)

set(DET_SOURCES src/coco.cpp src/yolo_common.cpp src/detectors.cpp src/lane_segmenter.cpp)
set(DET_DEFS "")
set(DET_LIBS "")

# ---- optional: ONNX Runtime (PC inference / offline JSON generation) ----
set(ONNXRUNTIME_ROOT "$ENV{HOME}/.local/onnxruntime" CACHE PATH "ONNX Runtime install (include/ + lib/)")
find_path(ORT_INCLUDE onnxruntime_cxx_api.h PATHS ${ONNXRUNTIME_ROOT}/include ${ONNXRUNTIME_ROOT}/include/onnxruntime NO_DEFAULT_PATH)
find_library(ORT_LIB onnxruntime PATHS ${ONNXRUNTIME_ROOT}/lib NO_DEFAULT_PATH)
if(ORT_INCLUDE AND ORT_LIB)
  message(STATUS "object_detection: ONNX Runtime backend ON (${ORT_LIB})")
  list(APPEND DET_SOURCES src/onnxruntime_detector.cpp src/onnxruntime_lane.cpp)
  list(APPEND DET_DEFS HAVE_ONNXRUNTIME)
  list(APPEND DET_LIBS ${ORT_LIB})
else()
  message(STATUS "object_detection: ONNX Runtime not found -> backend OFF")
endif()

# ---- optional: TensorRT (Jetson Orin) ----
find_package(CUDAToolkit QUIET)
find_path(TRT_INCLUDE NvInfer.h PATHS /usr/include/aarch64-linux-gnu /usr/include/x86_64-linux-gnu /usr/local/tensorrt/include)
find_library(TRT_NVINFER nvinfer PATHS /usr/lib/aarch64-linux-gnu /usr/lib/x86_64-linux-gnu /usr/local/tensorrt/lib)
if(CUDAToolkit_FOUND AND TRT_INCLUDE AND TRT_NVINFER)
  message(STATUS "object_detection: TensorRT backend ON (${TRT_NVINFER})")
  list(APPEND DET_SOURCES src/tensorrt_detector.cpp src/tensorrt_lane.cpp)
  list(APPEND DET_DEFS HAVE_TENSORRT)
  list(APPEND DET_LIBS ${TRT_NVINFER} CUDA::cudart)
else()
  message(STATUS "object_detection: TensorRT not found -> backend OFF")
endif()

add_library(object_detection_core SHARED ${DET_SOURCES})
target_include_directories(object_detection_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(ORT_INCLUDE)
  target_include_directories(object_detection_core PRIVATE ${ORT_INCLUDE})
endif()
if(TRT_INCLUDE)
  target_include_directories(object_detection_core PRIVATE ${TRT_INCLUDE})
endif()
target_compile_definitions(object_detection_core PUBLIC ${DET_DEFS})
target_link_libraries(object_detection_core PUBLIC ${OpenCV_LIBS} Eigen3::Eigen fusion_common::fusion_common ${DET_LIBS})
set_target_properties(object_detection_core PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)

add_executable(object_detection_node src/object_detection_node.cpp)
target_link_libraries(object_detection_node object_detection_core)
ament_target_dependencies(object_detection_node rclcpp sensor_msgs perception_msgs)

add_executable(bag_to_detections_json src/bag_to_detections_json.cpp)
target_link_libraries(bag_to_detections_json object_detection_core)
ament_target_dependencies(bag_to_detections_json rclcpp sensor_msgs rosbag2_cpp)

set_target_properties(object_detection_node bag_to_detections_json PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
install(TARGETS object_detection_node bag_to_detections_json DESTINATION lib/${PROJECT_NAME})

# ---- optional: lane_viewer (offline display of the lane / road chain on a bag; needs highgui) ----
find_package(nav_msgs QUIET)
find_package(lane_estimation QUIET)
get_filename_component(_cv_prefix "${OpenCV_DIR}/../../.." ABSOLUTE)
find_library(CV_HIGHGUI NAMES opencv_highgui HINTS ${_cv_prefix}/lib ${_cv_prefix}/lib64 NO_DEFAULT_PATH)
find_library(CV_VIDEOIO NAMES opencv_videoio HINTS ${_cv_prefix}/lib ${_cv_prefix}/lib64 NO_DEFAULT_PATH)
if(nav_msgs_FOUND AND lane_estimation_FOUND AND (TARGET opencv_highgui OR CV_HIGHGUI) AND
   (TARGET opencv_videoio OR CV_VIDEOIO))
  message(STATUS "object_detection: lane_viewer ON")
  fusion_opencv_libs(VIEWER_CV_LIBS opencv_highgui opencv_videoio)
  add_executable(lane_viewer src/lane_viewer.cpp)
  target_link_libraries(lane_viewer object_detection_core lane_estimation::lane_estimation ${VIEWER_CV_LIBS})
  ament_target_dependencies(lane_viewer rclcpp sensor_msgs nav_msgs rosbag2_cpp)
  set_target_properties(lane_viewer PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
  install(TARGETS lane_viewer DESTINATION lib/${PROJECT_NAME})
else()
  message(STATUS "object_detection: lane_viewer OFF (needs nav_msgs, lane_estimation, OpenCV highgui + videoio)")
endif()

install(TARGETS object_detection_core ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})

ament_package()
