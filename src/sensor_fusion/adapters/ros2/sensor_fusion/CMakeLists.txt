cmake_minimum_required(VERSION 3.8)
project(sensor_fusion)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra)

# Stages compiled into the node (a stage that is OFF is not in the binary):
#   SF_WITH_IMM        radar.tracker imm (raw radar -> IMM tracker in this node); OFF: tracks only from
#                      the EKF tracker node (radar.tracker ekf)
#   SF_WITH_BYTETRACK  ByteTrack on the detector boxes; OFF: the camera objects must already carry
#                      track ids (a camera tracker upstream)
#   SF_WITH_LANE       lane / road boundaries from the camera segmentation mask (lane.topic)
option(SF_WITH_IMM "IMM radar tracker (radar.tracker imm)" ON)
option(SF_WITH_BYTETRACK "ByteTrack 2D tracking of the detector boxes" ON)
option(SF_WITH_LANE "lane / road boundaries from the camera segmentation" ON)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(perception_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(sensor_fusion_core REQUIRED)

add_executable(sensor_fusion_node src/sensor_fusion_node.cpp)
target_link_libraries(sensor_fusion_node sensor_fusion_core::sensor_fusion_core)
if(SF_WITH_IMM)
  find_package(imm_tracker REQUIRED)
  target_link_libraries(sensor_fusion_node imm_tracker::imm_tracker)
  target_compile_definitions(sensor_fusion_node PRIVATE SF_WITH_IMM)
endif()
if(SF_WITH_BYTETRACK)
  find_package(byte_tracker REQUIRED)
  target_link_libraries(sensor_fusion_node byte_tracker::byte_tracker)
  target_compile_definitions(sensor_fusion_node PRIVATE SF_WITH_BYTETRACK)
endif()
if(SF_WITH_LANE)
  find_package(lane_estimation REQUIRED)
  target_link_libraries(sensor_fusion_node lane_estimation::lane_estimation)
  target_compile_definitions(sensor_fusion_node PRIVATE SF_WITH_LANE)
endif()
message(STATUS "sensor_fusion_node: IMM tracker ${SF_WITH_IMM}, ByteTrack ${SF_WITH_BYTETRACK}, lane ${SF_WITH_LANE}")
ament_target_dependencies(sensor_fusion_node rclcpp sensor_msgs nav_msgs std_msgs geometry_msgs perception_msgs tf2 tf2_ros)

install(TARGETS sensor_fusion_node DESTINATION lib/${PROJECT_NAME})

ament_package()
