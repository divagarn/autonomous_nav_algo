cmake_minimum_required(VERSION 3.8)
project(sensor_fusion_tests)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra)

# Unit tests of the core modules (no ROS): colcon test --packages-select sensor_fusion_tests
find_package(sensor_fusion_core REQUIRED)
find_package(imm_tracker REQUIRED)
find_package(byte_tracker REQUIRED)
find_package(lane_estimation REQUIRED)

include(CTest)
if(NOT BUILD_TESTING)
  return()
endif()
find_package(GTest QUIET)
if(NOT GTest_FOUND AND EXISTS /usr/src/googletest/CMakeLists.txt)
  add_subdirectory(/usr/src/googletest ${CMAKE_CURRENT_BINARY_DIR}/googletest EXCLUDE_FROM_ALL)
  add_library(GTest::gtest_main ALIAS gtest_main)
endif()
if(NOT TARGET GTest::gtest_main)
  message(WARNING "sensor_fusion_tests: GoogleTest not found (sudo apt install libgtest-dev), tests not built")
  return()
endif()
include(GoogleTest)
add_executable(test_fusion_core test_fusion_core.cpp)
target_link_libraries(test_fusion_core sensor_fusion_core::sensor_fusion_core imm_tracker::imm_tracker
  byte_tracker::byte_tracker GTest::gtest_main)
gtest_discover_tests(test_fusion_core)
add_executable(test_byte_tracker test_byte_tracker.cpp)
target_link_libraries(test_byte_tracker byte_tracker::byte_tracker GTest::gtest_main)
gtest_discover_tests(test_byte_tracker)
add_executable(test_lane_estimator test_lane_estimator.cpp)
target_link_libraries(test_lane_estimator lane_estimation::lane_estimation GTest::gtest_main)
gtest_discover_tests(test_lane_estimator)
add_executable(test_ground_height test_ground_height.cpp)
target_link_libraries(test_ground_height fusion_common::fusion_common GTest::gtest_main)
gtest_discover_tests(test_ground_height)
add_executable(test_path_tracker test_path_tracker.cpp)
target_link_libraries(test_path_tracker lane_estimation::lane_estimation GTest::gtest_main)
gtest_discover_tests(test_path_tracker)
